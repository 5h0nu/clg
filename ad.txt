#include <stdio.h>

int main() {
    int n, m, i, j, currentWeight = 0;
    printf("Enter number of objects, weights, profits, and max capacity:\n");
    if (scanf("%d", &n) != 1) return 1;

    int w[n], p[n]; // Using Variable Length Arrays (VLA) supported since C99
    double x[n], ratio[n], maxprofit = 0.0;

    for (i = 0; i < n; i++) x[i] = 0.0;
    for (i = 0; i < n; i++) scanf("%d", &w[i]);
    for (i = 0; i < n; i++) {
        scanf("%d", &p[i]);
        ratio[i] = (double)p[i] / w[i];
    }
    scanf("%d", &m);

    // Bubble sort by profit/weight ratio
    for (i = 0; i < n - 1; i++) {
        for (j = i + 1; j < n; j++) {
            if (ratio[i] < ratio[j]) {
                double t1 = ratio[i]; ratio[i] = ratio[j]; ratio[j] = t1;
                int t2 = w[i]; w[i] = w[j]; w[j] = t2;
                t2 = p[i]; p[i] = p[j]; p[j] = t2;
            }
        }
    }

    // Greedy Knapsack calculation
    for (i = 0; i < n; i++) {
        if (currentWeight + w[i] <= m) {
            x[i] = 1.0;
            currentWeight += w[i];
            maxprofit += p[i];
        } else {
            x[i] = (double)(m - currentWeight) / w[i];
            maxprofit += x[i] * p[i];
            break;
        }
    }

    printf("\nOptimal solution: %.2f\nSolution vector: ", maxprofit);
    for (i = 0; i < n; i++) printf("%.2f\t", x[i]);
    printf("\n");
    return 0;
}

#include <stdio.h>
#include <sys/time.h>

void selsort(int a[], int n) {
    for (int j = 0; j < n - 1; j++) {
        int small = a[j], pos = j;
        for (int i = j + 1; i < n; i++)
            if (a[i] < small) { small = a[i]; pos = i; }
        int temp = a[j]; a[j] = a[pos]; a[pos] = temp;
    }
}

int main() {
    int n, a[10000];
    struct timeval start, end;

    printf("\nEnter the n value: ");
    if (scanf("%d", &n) != 1) return 1;
    if (n > 10000) { printf("Array size too large, reducing it to 10000.\n"); n = 10000; }

    printf("\nEnter the array: ");
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    gettimeofday(&start, NULL);
    selsort(a, n);
    gettimeofday(&end, NULL);

    long sec = end.tv_sec - start.tv_sec;
    long usec = end.tv_usec - start.tv_usec;
    if (usec < 0) { sec--; usec += 1000000; }

    printf("\nTime taken is: %ld seconds and %ld microseconds\n\nSorted array is: ", sec, usec);
    for (int i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");
    return 0;
}

#include <stdio.h>

#include <sys/time.h>
#include <time.h>

void Exch(int *p, int *q) { int t = *p; *p = *q; *q = t; }

void QuickSort(int a[], int low, int high) {
    if (low >= high) return;
    int i = low + 1, j = high, key = low;
    while (i <= j) {
        while (a[i] <= a[key] && i <= high) i++;
        while (a[j] > a[key] && j >= low) j--;
        if (i < j) Exch(&a[i], &a[j]);
    }
    Exch(&a[j], &a[key]);
    QuickSort(a, low, j - 1);
    QuickSort(a, j + 1, high);
}

int main() {
    int n, k;
    struct timeval start, end;
    
    printf("\nEnter how many numbers: ");
    if (scanf("%d", &n) != 1) return 1;

#include <stdio.h>

#include <sys/time.h>

void Merge(int a[], int low, int mid, int high) {
    int i = low, j = mid + 1, k = 0, b[high - low + 1];
    while (i <= mid && j <= high) b[k++] = (a[i] <= a[j]) ? a[i++] : a[j++];
    while (i <= mid) b[k++] = a[i++];
    while (j <= high) b[k++] = a[j++];
    for (i = low, k = 0; i <= high; i++, k++) a[i] = b[k];
}

void MergeSort(int a[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;
        MergeSort(a, low, mid);
        MergeSort(a, mid + 1, high);
        Merge(a, low, mid, high);
    }
}

#include <stdio.h>
#include <stdlib.h>

int a[30], count = 0;

int place(int pos) {
    for (int i = 1; i < pos; i++)
        if (a[i] == a[pos] || abs(a[i] - a[pos]) == abs(i - pos)) return 0;
    return 1;
}

void print_sol(int n) {
    printf("\n\nSolution #%d:\n", ++count);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) printf("%s", (a[i] == j) ? "Q\t" : "*\t");
        printf("\n");
    }
}

void queen(int n) {
    int k = 1; a[k] = 0;
    while (k > 0) {
        a[k]++;
        while (a[k] <= n && !place(k)) a[k]++;
        if (a[k] <= n) {
            if (k == n) print_sol(n);
            else a[++k] = 0;
        } else k--;
    }
}

int main() {
    int n;
    printf("Enter the number of Queens: ");
    if (scanf("%d", &n) != 1) return 1;
    queen(n);
    printf("\nTotal solutions = %d\n", count);
    return 0;
}
